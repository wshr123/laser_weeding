# 双振镜系统改造策略

## 一、现状分析

### 1.1 下位机现状
- ✅ 已支持多振镜实例：`XY2_100_multi`库支持最多2个振镜（MAX_GALVO_INSTANCES = 2）
- ✅ 硬件资源：使用FTM1和FTM2两个定时器模块分别控制两个振镜
- ❌ 当前Arduino代码（`control_tennsy.ino`）只实例化了一个振镜对象
- ❌ 串口协议未区分振镜ID

### 1.2 上位机现状
- ❌ `send_to_teensy.py`：只支持单个振镜控制器
- ❌ `main.py`：只有一个振镜控制线程
- ❌ 配置文件：只有一套振镜参数

## 二、改造策略

### 2.1 串口通信协议扩展

#### 方案A：命令前缀方式（推荐）
```
# 振镜0（第一个振镜）
GALVO0:XY:x,y        # 移动振镜0到位置(x,y)
GALVO0:LASER:ON      # 打开振镜0的激光
GALVO0:LASER:OFF     # 关闭振镜0的激光
GALVO0:CENTER        # 振镜0回到中心
GALVO0:STATUS        # 查询振镜0状态

# 振镜1（第二个振镜）
GALVO1:XY:x,y
GALVO1:LASER:ON
GALVO1:LASER:OFF
GALVO1:CENTER
GALVO1:STATUS

# 全局命令（紧急控制）
STOP                 # 紧急停止所有振镜（同时关闭所有激光）

# 注意：没有全局LASER:ON/OFF命令，每个振镜的激光必须独立控制
```

#### 方案B：命令编号方式
```
XY0:x,y              # 振镜0
XY1:x,y              # 振镜1
LASER0:ON            # 振镜0激光
LASER1:ON            # 振镜1激光
```

**推荐方案A**，因为：
- 更清晰易读
- 便于扩展更多振镜
- 保持向后兼容（无前缀时默认控制振镜0）

void parseCommand(String cmd) {
    cmd.toUpperCase();
    
    // 解析振镜ID
    uint8_t galvo_id = 0;  // 默认振镜0
    bool has_galvo_prefix = false;
    
    if (cmd.startsWith("GALVO0:")) {
        galvo_id = 0;
        has_galvo_prefix = true;
        cmd = cmd.substring(7);  // 移除"GALVO0:"
    } else if (cmd.startsWith("GALVO1:")) {
        galvo_id = 1;
        has_galvo_prefix = true;
        cmd = cmd.substring(7);  // 移除"GALVO1:"
    }
    
    // XY位置命令
    if (cmd.startsWith("XY:")) {
        // ... 解析x,y ...
        if (galvo_id == 0) {
            galvo0.setSignedXY(x, y);
        } else {
            galvo1.setSignedXY(x, y);
        }
    }
    
    // 激光控制（必须指定振镜ID）
    if (cmd == "LASER:ON") {
        if (has_galvo_prefix) {
            // 控制指定振镜的激光
            setLaser(galvo_id, true);
        } else {
            // 无前缀的LASER:ON命令无效，返回错误
            sendResponse("ERROR:LASER command requires GALVO prefix");
        }
    }
    else if (cmd == "LASER:OFF") {
        if (has_galvo_prefix) {
            setLaser(galvo_id, false);
        } else {
            sendResponse("ERROR:LASER command requires GALVO prefix");
        }
    }
    // 紧急停止（全局命令，关闭所有激光）
    else if (cmd == "STOP") {
        setLaser(0, false);
        setLaser(1, false);
        // ... 停止所有运动 ...
    }
    // ... 其他命令 ...
}

### 2.3 上位机Python代码改造

#### 2.3.1 `send_to_teensy.py` 改造

**方案：创建多振镜控制器类**

```python
class MultiGalvoController:
    """多振镜控制器 - 管理多个振镜实例"""
    
    def __init__(self, port='/dev/ttyACM0', baudrate=115200, num_galvos=2):
        self.port = port
        self.baudrate = baudrate
        self.num_galvos = num_galvos
        self.serial_port = None
        
        # 每个振镜的状态
        self.galvo_states = []
        for i in range(num_galvos):
            self.galvo_states.append({
                'current_x': 0,
                'current_y': 0,
                'laser_enabled': False
            })
        
        self.connect()
    
    def move_to_position(self, galvo_id, x, y):
        """移动指定振镜到位置"""
        if galvo_id >= self.num_galvos:
            rospy.logerr(f"Invalid galvo_id: {galvo_id}")
            return False
        
        x = max(-32767, min(32767, int(x)))
        y = max(-32767, min(32767, int(y)))
        
        command = f"GALVO{galvo_id}:XY:{x},{y}"
        if self.send_command(command):
            self.galvo_states[galvo_id]['current_x'] = y
            self.galvo_states[galvo_id]['current_y'] = x
            return True
        return False
    
    def set_laser(self, galvo_id, enable):
        """控制指定振镜的激光"""
        if galvo_id >= self.num_galvos:
            return False
        
        command = f"GALVO{galvo_id}:LASER:{'ON' if enable else 'OFF'}"
        if self.send_command(command):
            self.galvo_states[galvo_id]['laser_enabled'] = enable
            return True
        return False
    
    def emergency_stop(self):
        """紧急停止所有振镜（关闭所有激光）"""
        command = "STOP"
        return self.send_command(command)
```

2. **目标分配策略（使用策略1和策略3）**

```python
def calculate_galvo_scan_region(self, galvo_id, work_height_mm):
    """
    根据振镜扫描角度和工作高度，计算振镜在图像中的扫描区域
    
    关键参数：
    - 振镜扫描角度（从配置文件读取）
    - 工作高度（可配置参数，单位：mm）
    - 相机内参（用于像素到物理尺寸转换）
    """
    transform = self.coordinate_transforms[galvo_id]
    galvo_params = transform.params['galvo_params']
    
    # 获取振镜扫描角度（度转弧度）
    scan_angle_rad = np.radians(galvo_params['scan_angle'] / 2.0)
    
    # 计算扫描半径（在工作高度平面上的实际距离，单位：mm）
    scan_radius_mm = work_height_mm * np.tan(scan_angle_rad)
    
    # 将振镜中心位置（码值0,0）映射到像素坐标
    center_pixel = transform.galvo_code_to_pixel(
        0, 0, self.image_width, self.image_height
    )
    
    # 计算扫描半径对应的像素距离
    fx = transform.params['camera_matrix']['fx']
    pixel_per_mm = fx / (work_height_mm / 1000.0)
    scan_radius_pixel = scan_radius_mm * pixel_per_mm / 1000.0
    
    return center_pixel, scan_radius_pixel

def assign_target_to_galvo(self, target_center, work_height_mm=1000.0):
    """
    综合目标分配策略：策略1（扫描范围）+ 策略3（距离优先）
    
    逻辑：
    1. 计算每个振镜的扫描区域（基于扫描角度和工作高度）
    2. 检查目标在哪个振镜的扫描范围内
    3. 如果目标在多个振镜范围内，选择距离振镜中心最近的
    4. 如果目标不在任何振镜范围内，选择距离最近的作为备选
    """
    in_range_galvos = []
    distances = []
    
    for galvo_id in range(self.num_galvos):
        center_pixel, scan_radius = self.calculate_galvo_scan_region(
            galvo_id, work_height_mm
        )
        
        if center_pixel is None:
            continue
        
        # 计算目标到振镜中心的距离
        dist = np.hypot(
            target_center[0] - center_pixel[0],
            target_center[1] - center_pixel[1]
        )
        distances.append((dist, galvo_id))
        
        # 检查是否在扫描范围内
        if dist <= scan_radius:
            in_range_galvos.append((dist, galvo_id))
    
    # 优先选择扫描范围内的振镜，按距离排序
    if in_range_galvos:
        in_range_galvos.sort(key=lambda x: x[0])
        return in_range_galvos[0][1]
    
    # 如果都不在范围内，选择距离最近的
    if distances:
        distances.sort(key=lambda x: x[0])
        return distances[0][1]
    
    return 0  # 默认返回振镜0
```

## 四、目标分配策略详解

### 策略1：按扫描范围分配（主要策略）

根据振镜的扫描角度和工作高度，计算每个振镜在图像中的实际扫描区域，优先分配给扫描范围内的振镜。

```python
def calculate_galvo_scan_region(self, galvo_id, work_height_mm):
    """
    根据振镜扫描角度和工作高度，计算振镜在图像中的扫描区域
    
    Args:
        galvo_id: 振镜ID (0或1)
        work_height_mm: 工作高度（毫米）
    
    Returns:
        (center_pixel, scan_radius_pixel): 扫描中心像素坐标和扫描半径（像素）
    """
    transform = self.coordinate_transforms[galvo_id]
    galvo_params = transform.params['galvo_params']
    
    # 获取振镜扫描角度（度转弧度）
    scan_angle_rad = np.radians(galvo_params['scan_angle'] / 2.0)  # 半角
    
    # 计算扫描半径（在工作高度平面上的实际距离，单位：mm）
    scan_radius_mm = work_height_mm * np.tan(scan_angle_rad)
    
    # 将振镜中心位置（码值0,0）映射到像素坐标
    galvo_center_code = (0, 0)
    center_pixel = transform.galvo_code_to_pixel(
        galvo_center_code[0], galvo_center_code[1],
        self.image_width, self.image_height
    )
    
    if center_pixel is None:
        return None, None
    
    # 计算扫描半径对应的像素距离
    # 需要根据相机内参和深度信息转换
    # 简化：使用像素/mm比例（从相机内参计算）
    fx = transform.params['camera_matrix']['fx']
    pixel_per_mm = fx / (work_height_mm / 1000.0)  # 假设工作高度为米
    scan_radius_pixel = scan_radius_mm * pixel_per_mm / 1000.0
    
    return center_pixel, scan_radius_pixel

def assign_by_scan_range(self, target_center, work_height_mm=1000.0):
    """
    检查目标在哪个振镜的扫描范围内
    
    策略：
    1. 首先检查目标是否在振镜的扫描范围内
    2. 如果都在范围内，选择距离振镜中心最近的
    3. 如果都不在范围内，选择距离振镜中心最近的（作为备选）
    """
    in_range_galvos = []
    distances = []
    
    for galvo_id in range(self.num_galvos):
        # 计算振镜扫描区域
        center_pixel, scan_radius = self.calculate_galvo_scan_region(
            galvo_id, work_height_mm
        )
        
        if center_pixel is None:
            continue
        
        # 计算目标到振镜中心的距离
        dist = np.hypot(
            target_center[0] - center_pixel[0],
            target_center[1] - center_pixel[1]
        )
        distances.append((dist, galvo_id))
        
        # 检查是否在扫描范围内
        if dist <= scan_radius:
            in_range_galvos.append((dist, galvo_id))
    
    # 策略1：优先选择扫描范围内的振镜
    if in_range_galvos:
        # 在范围内的振镜中，选择距离最近的
        in_range_galvos.sort(key=lambda x: x[0])
        return in_range_galvos[0][1]
    
    # 策略2：如果都不在范围内，选择距离最近的（作为备选）
    if distances:
        distances.sort(key=lambda x: x[0])
        return distances[0][1]
    
    # 默认返回振镜0
    return 0
```

### 策略2：负载均衡（可选辅助策略）
```python
def assign_by_load_balance(self, target_center):
    """选择当前处理目标最少的振镜（作为辅助策略）"""
    galvo0_load = sum(1 for tid, gid in self.target_galvo_map.items() 
                      if gid == 0 and tid not in self.processed_targets)
    galvo1_load = sum(1 for tid, gid in self.target_galvo_map.items() 
                      if gid == 1 and tid not in self.processed_targets)
    
    if galvo0_load <= galvo1_load:
        return 0
    else:
        return 1
```

### 策略3：距离优先（与策略1结合使用）

在策略1中已经实现了距离优先的逻辑：
- 如果目标在多个振镜的扫描范围内，选择距离最近的
- 如果目标不在任何振镜的扫描范围内，选择距离最近的作为备选

### 综合分配策略（推荐实现）

```python
def assign_target_to_galvo(self, target_center, work_height_mm=1000.0):
    """
    综合目标分配策略
    
    主要使用策略1（扫描范围）+ 策略3（距离优先）
    可选使用策略2（负载均衡）作为辅助
    
    Args:
        target_center: 目标中心像素坐标 [x, y]
        work_height_mm: 工作高度（毫米），用于计算扫描区域
    
    Returns:
        galvo_id: 分配的振镜ID
    """
    # 策略1：按扫描范围分配（主要策略）
    galvo_id = self.assign_by_scan_range(target_center, work_height_mm)
    
    # 可选：如果两个振镜负载差异很大，可以调整分配
    # 这里暂时不使用负载均衡，保持简单
    
    return galvo_id
```

## 五、注意事项

### 5.1 硬件要求
- 确保Teensy有足够的GPIO引脚用于两个振镜（每个振镜需要8个数据引脚）
- 确保有两个独立的激光控制引脚
- 检查FTM1和FTM2定时器资源可用

### 5.2 软件兼容性
- **激光控制必须指定振镜ID**：没有全局LASER:ON/OFF命令，每个振镜的激光必须独立控制
- 配置文件支持单振镜和多振镜两种模式
- Launch文件支持参数切换
- 紧急停止命令（STOP）会同时关闭所有激光

### 5.3 性能考虑
- 两个振镜控制线程独立运行，互不干扰
- 目标分配策略要高效，避免频繁切换
- 串口通信要考虑命令队列，避免阻塞

### 5.4 安全考虑
- 紧急停止命令要同时停止两个振镜
- 激光超时保护要分别监控
- 位置限制要分别检查

## 八、总结

改造核心思路：
1. **串口协议扩展**：添加振镜ID前缀，激光控制必须指定振镜ID
2. **下位机多实例**：创建两个XY2_100对象
3. **上位机多控制器**：支持管理多个振镜
4. **目标智能分配**：
   - **策略1（主要）**：根据振镜扫描角度和工作高度计算扫描区域，优先分配给扫描范围内的振镜
   - **策略3（辅助）**：在扫描范围内的振镜中，选择距离振镜中心最近的
5. **独立控制线程**：每个振镜独立的高频控制循环

这样改造后，系统可以：
- ✅ 同时控制两个独立的振镜系统
- ✅ 根据实际扫描区域智能分配目标（而非简单的X坐标分界）
- ✅ 每个振镜的激光独立控制，提高安全性
- ✅ 提高系统处理能力和覆盖范围
- ✅ 基于物理模型的精确目标分配
